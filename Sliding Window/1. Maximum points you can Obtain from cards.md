# [1. Maximum points you can Obtain from cards](https://leetcode.com/problems/maximum-points-you-can-obtain-from-cards/description/)

```cpp
class Solution {

public:

    int maxScore(vector<int>& arr, int k) {

        int l=k-1, r = arr.size()-1;

        int n= arr.size();

  
  

       int ans= accumulate(arr.begin() , arr.begin() +k ,0);

       int sum = ans;

  

        for(l = k-1; l >=0; l--)

        {

            sum -= arr[l];

            sum+= arr[r];

            r--;

            ans = max(ans , sum);

        }

  

        return ans;

    }

};
```


**Time Complexity**: `O(K)`

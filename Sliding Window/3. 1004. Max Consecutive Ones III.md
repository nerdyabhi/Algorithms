# [1004. Max Consecutive Ones III](https://leetcode.com/problems/max-consecutive-ones-iii/)



# Better Approach
```cpp
class Solution {
public:
    int longestOnes(vector<int>& nums, int k) {
        // finding the longest subarray with atmost k zeroes
        int zeroes = 0 , i =0 , ans=0;

        for(int j=0; j<nums.size(); j++)
        {
            if(nums[j] ==0) zeroes++;
            if(zeroes >k)
            {
               while(nums[i] !=0) i++;
               zeroes--;
               i++;
            }

            ans = max(ans , j-i+1);
        }

        return ans;
    }
};
```
